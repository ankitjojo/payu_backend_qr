FLOW OVERVIEW
=============
1.  Frontend (views/index.ejs + public/endpoints.js + public/logic.js)
    - User fills mandate form and clicks "Generate QR Mandate"
    - logic.js calls POST /api/payments/generate-qr with JSON body

2.  Backend Route (src/api/routes/payment.routes.ts)
    - Route: POST /api/payments/generate-qr  →  PaymentController.generateQR

3.  Controller (src/api/controllers/payment.controller.ts)
    - PaymentController.generateQR → PaymentService.generateQRMandate

4.  Service Layer (src/services/payment.service.ts)
    - PaymentService.generateQRMandate → PayUService.generateQRMandateS2S

5.  PayU Service (src/services/payu.service.ts)
    - Builds payload: pg=UPI, bankcode=INTENT, txn_s2s_flow=4, api_version=7
    - Calls generateHashForPayment (includes si_details in hash)
    - POSTs URL-encoded form to payuConfig.endpoints.paymentUrl via HttpHelper

6.  Hash Utility (src/utils/hash.util.ts)
    - generateHashForPayment: key|txnid|amount|productinfo|firstname|email|...|si_details|SALT

7.  HTTP Utility (src/utils/http.util.ts)
    - HttpHelper.postForm → axios POST with Content-Type: application/x-www-form-urlencoded

8.  PayU Response
    - Returns JSON with result.intentURIData (the UPI deep-link)

9.  Frontend QR Rendering (public/logic.js)
    - Extracts intentURIData from response
    - Uses QRCode.js (CDN) to render QR image client-side via QRCode.toDataURL()
    - Sets <img id="qrImage">.src = base64 data URL — no backend image serving needed

SUCCESS / FAILURE CALLBACKS
===========================
- PayU POSTs to POST /api/payments/success  →  PaymentController.handleSuccess
- PayU POSTs to POST /api/payments/failure  →  PaymentController.handleFailure
- Both validate reverse hash and log to payment_logs.json

import crypto from 'crypto';
import { payuConfig } from '../config/payu.config';

/**
 * Generates a SHA-512 hash string.
 * @param hashString The string to hash
 * @returns The hashed string
 */
export const generateHash = (hashString: string): string => {
  const hash = crypto.createHash('sha512');
  hash.update(hashString);
  return hash.digest('hex');
};

/**
 * Generates the hash required for initiating a PayU transaction (including Zion Mandate).
 * The sequence for payment hash is: key|txnid|amount|productinfo|firstname|email|udf1|udf2|udf3|udf4|udf5||||||SALT
 */
export const generateHashForPayment = (
  txnid: string,
  amount: string,
  productinfo: string,
  firstname: string,
  email: string,
  si_details: string = '',
  udf1: string = '',
  udf2: string = '',
  udf3: string = '',
  udf4: string = '',
  udf5: string = '',
): string => {
  const { merchantKey, salt } = payuConfig;
  if (si_details) {
      const hashString = `${merchantKey}|${txnid}|${amount}|${productinfo}|${firstname}|${email}|${udf1}|${udf2}|${udf3}|${udf4}|${udf5}||||||${si_details}|${salt}`;
      return generateHash(hashString);
  } else {
      const hashString = `${merchantKey}|${txnid}|${amount}|${productinfo}|${firstname}|${email}|${udf1}|${udf2}|${udf3}|${udf4}|${udf5}||||||${salt}`;
      return generateHash(hashString);
  }
};

/**
 * Validates the hash received from PayU (reverse hash).
 * The sequence for reverse hash is: SALT|status||||||udf5|udf4|udf3|udf2|udf1|email|firstname|productinfo|amount|txnid|key
 */
export const validateReverseHash = (
  status: string,
  txnid: string,
  amount: string,
  productinfo: string,
  firstname: string,
  email: string,
  receivedHash: string,
  udf1: string = '',
  udf2: string = '',
  udf3: string = '',
  udf4: string = '',
  udf5: string = '',
): boolean => {
  const { merchantKey, salt } = payuConfig;
  const hashString = `${salt}|${status}||||||${udf5}|${udf4}|${udf3}|${udf2}|${udf1}|${email}|${firstname}|${productinfo}|${amount}|${txnid}|${merchantKey}`;
  const calculatedHash = generateHash(hashString);
  return calculatedHash === receivedHash;
};

/**
 * Generates hash for generic API commands (like verify_payment).
 * Formula: key|command|var1|salt
 */
export const generateHashForCommand = (command: string, var1: string): string => {
  const { merchantKey, salt } = payuConfig;
  const hashString = `${merchantKey}|${command}|${var1}|${salt}`;
  return generateHash(hashString);
};

import axios, { AxiosRequestConfig, AxiosResponse } from 'axios';
import { logger } from './logger.util';

/**
 * Common HTTP helper class for making external API requests.
 */
export class HttpHelper {
  /**
   * Generic POST request method
   */
  public static async post<T>(url: string, data: any, config?: AxiosRequestConfig): Promise<T> {
    try {
      logger.info(`Making POST request to ${url}`);
      const response: AxiosResponse<T> = await axios.post(url, data, config);
      
      const responseDataToLog = typeof response.data === 'string' 
        ? response.data.substring(0, 1000) + ((response.data as string).length > 1000 ? '...' : '')
        : JSON.stringify(response.data, null, 2);
        
      logger.info(`Response from ${url}:`, responseDataToLog);
      return response.data;
    } catch (error: any) {
      logger.error(`POST request failed for ${url}:`, error.message);
      if (error.response) {
        const errorDataToLog = typeof error.response.data === 'string'
          ? error.response.data.substring(0, 1000)
          : JSON.stringify(error.response.data, null, 2);
        logger.error(`Error Response from ${url}:`, errorDataToLog);
      }
      throw error;
    }
  }

  /**
   * Specific POST request tailored for PayU URL-encoded forms
   */
  public static async postForm<T>(url: string, data: Record<string, string>): Promise<T> {
    logger.info(`[postForm] Payload to ${url}:`, JSON.stringify(data, null, 2));

    const params = new URLSearchParams();
    for (const key in data) {
      if (data.hasOwnProperty(key)) {
        params.append(key, String(data[key]));
      }
    }

    const config: AxiosRequestConfig = {
      headers: {
        'Accept': 'application/json',
        'Content-Type': 'application/x-www-form-urlencoded',
      },
    };

    return this.post<T>(url, params, config);
  }
}

import { Request, Response, NextFunction } from 'express';
import { PaymentService } from '../../services/payment.service';
import { validateReverseHash } from '../../utils/hash.util';
import { logger } from '../../utils/logger.util';
import fs from 'fs';
import path from 'path';
import { config } from '../../config';

const appendLogToFile = (data: any, type: 'success' | 'failure') => {
  const logFilePath = path.resolve(process.cwd(), 'payment_logs.json');
  const logEntry = {
    timestamp: new Date().toISOString(),
    type,
    data
  };

  try {
    let logs: any[] = [];
    if (fs.existsSync(logFilePath)) {
      const fileContent = fs.readFileSync(logFilePath, 'utf8');
      if (fileContent) {
        logs = JSON.parse(fileContent);
      }
    }
    logs.push(logEntry);
    fs.writeFileSync(logFilePath, JSON.stringify(logs, null, 2));
  } catch (err) {
    logger.error('Failed to write log to file: ' + err);
  }
};

export class PaymentController {
  private paymentService: PaymentService;

  constructor() {
    this.paymentService = new PaymentService();
  }

  public initiateMandate = async (req: Request, res: Response, next: NextFunction) => {
    try {
      const { amount, productinfo, firstname, email, phone, siDetails } = req.body;

      const payload = this.paymentService.initiateMandate(
        amount,
        productinfo,
        firstname,
        email,
        phone,
        siDetails,
      );

      res.status(200).json({
        success: true,
        data: payload,
      });
    } catch (error) {
      next(error);
    }
  };

  public generateQR = async (req: Request, res: Response, next: NextFunction) => {
    try {
      const { amount, productinfo, firstname, email, phone, siDetails } = req.body;

      const payload = await this.paymentService.generateQRMandate(
        amount,
        productinfo,
        firstname,
        email,
        phone,
        siDetails,
      );

      res.status(200).json({
        success: true,
        data: payload,
      });
    } catch (error: any) {
      logger.error('Error generating QR Mandate: ' + error.message);
      res.status(500).json({
        success: false,
        error: error.message,
        payuData: error.response?.data,
      });
    }
  };

  public generateUpiCollect = async (req: Request, res: Response, next: NextFunction) => {
    try {
      const { amount, productinfo, firstname, email, phone, vpa, siDetails } = req.body;

      const payload = await this.paymentService.generateUpiCollectMandate(
        amount,
        productinfo,
        firstname,
        email,
        phone,
        vpa,
        siDetails,
      );

      res.status(200).json({
        success: true,
        data: payload,
      });
    } catch (error: any) {
      console.error(error);
      res.status(500).json({
        success: false,
        error: error.message,
        payuData: error.response?.data
      });
    }
  };

  public handleSuccess = async (req: Request, res: Response, next: NextFunction) => {
    try {
      const { status, txnid, amount, productinfo, firstname, email, hash } = req.body;

      logger.info(`Received success callback for txnid: ${txnid}`);

      // Optional: Verify the reverse hash to ensure the response is authentically from PayU
      const isValid = validateReverseHash(
        String(status),
        String(txnid),
        String(amount),
        String(productinfo),
        String(firstname),
        String(email),
        String(hash),
      );

      if (!isValid) {
        logger.warn(`Hash validation failed for txnid: ${txnid}`);
        return res.status(400).send('Hash validation failed');
      }

      // Normally: update database record for txnid to 'success'
      appendLogToFile(req.body, 'success');

      // Since this is a direct POST from PayU to our backend, we redirect the user
      res.redirect(`${config.checkoutUrl}/success`);
    } catch (error) {
      next(error);
    }
  };

  public handleFailure = async (req: Request, res: Response, next: NextFunction) => {
    try {
      const { txnid, error_Message } = req.body;
      logger.error(`Received failure callback for txnid: ${txnid}. Reason: ${error_Message}`);

      // Normally: update database record for txnid to 'failed'
      appendLogToFile(req.body, 'failure');

      res.redirect(`${config.checkoutUrl}/failure`);
    } catch (error) {
      next(error);
    }
  };

  public verifyPayment = async (req: Request, res: Response, next: NextFunction) => {
    try {
      const { txnid } = req.params;
      const result = await this.paymentService.verifyPayment(String(txnid));

      res.status(200).json({
        success: true,
        data: result,
      });
    } catch (error) {
      next(error);
    }
  };
}

document.addEventListener('DOMContentLoaded', () => {
    // Set default dates to today and 1 year from now
    const startDateInput = document.getElementById('paymentStartDate');
    const endDateInput = document.getElementById('paymentEndDate');
    
    const today = new Date();
    startDateInput.value = today.toISOString().split('T')[0];
    
    const nextYear = new Date();
    nextYear.setFullYear(today.getFullYear() + 1);
    endDateInput.value = nextYear.toISOString().split('T')[0];

    const form = document.getElementById('mandateForm');
    const responseContainer = document.getElementById('responseContainer');
    const jsonResponse = document.getElementById('jsonResponse');
    const submitBtn = document.getElementById('submitBtn');

    form.addEventListener('submit', async (e) => {
        e.preventDefault();
        
        submitBtn.disabled = true;
        submitBtn.textContent = 'Processing...';

        // Gather data
        const payload = {
            amount: document.getElementById('amount').value,
            productinfo: document.getElementById('productinfo').value,
            firstname: document.getElementById('firstname').value,
            email: document.getElementById('email').value,
            phone: document.getElementById('phone').value,
            siDetails: {
                billingAmount: document.getElementById('billingAmount').value,
                billingCurrency: 'INR',
                billingCycle: document.getElementById('billingCycle').value,
                billingInterval: 1, // Example fixed value
                paymentStartDate: document.getElementById('paymentStartDate').value,
                paymentEndDate: document.getElementById('paymentEndDate').value,
            }
        };

        try {
            // Call backend API
            const response = await fetch(endpoints.initiateMandate, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify(payload)
            });

            const result = await response.json();
            
            if (result.success) {
                // Show JSON response for debugging
                responseContainer.classList.remove('hidden');
                jsonResponse.textContent = JSON.stringify(result.data, null, 2);

                // Auto-submit form to PayU using the generated payload
                submitPayUForm(result.data);
            } else {
                alert('Failed to initiate mandate.');
            }
        } catch (error) {
            console.error('Error:', error);
            alert('An error occurred. Check console.');
        } finally {
            submitBtn.disabled = false;
            submitBtn.textContent = 'Initiate Mandate Setup';
        }
    });

    const qrBtn = document.getElementById('qrBtn');

    qrBtn.addEventListener('click', async (e) => {
        e.preventDefault();
        
        qrBtn.disabled = true;
        qrBtn.textContent = 'Processing...';

        // Gather data
        const payload = {
            amount: document.getElementById('amount').value,
            productinfo: document.getElementById('productinfo').value,
            firstname: document.getElementById('firstname').value,
            email: document.getElementById('email').value,
            phone: document.getElementById('phone').value,
            siDetails: {
                billingAmount: document.getElementById('billingAmount').value,
                billingCurrency: 'INR',
                billingCycle: document.getElementById('billingCycle').value,
                billingInterval: 1, // Example fixed value
                paymentStartDate: document.getElementById('paymentStartDate').value,
                paymentEndDate: document.getElementById('paymentEndDate').value,
            }
        };

        try {
            // Call backend API
            const response = await fetch(endpoints.generateQR, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json'
                },
                body: JSON.stringify(payload)
            });

            const result = await response.json();
            
            if (result.success) {
                // Show JSON response for debugging
                responseContainer.classList.remove('hidden');
                jsonResponse.textContent = JSON.stringify(result.data, null, 2);

                const payuResponse = result.data;
                const intentStr = (payuResponse && payuResponse.result) ? (payuResponse.result.intentURIData || payuResponse.result.intentUrl || payuResponse.result.qrString) : null;
                
                if (intentStr) {
                    const redirectText = document.getElementById('redirectText');
                    if (redirectText) redirectText.style.display = 'none';
                    
                    document.getElementById('qrCodeContainer').style.display = 'block';
                    
                    // Generate QR image securely and instantly on the frontend
                    QRCode.toDataURL(intentStr, { width: 250, margin: 2 }, function (err, url) {
                        if (err) {
                            console.error('QR Generation Error:', err);
                            return;
                        }
                        document.getElementById('qrImage').src = url;
                        document.getElementById('qrIntentText').textContent = intentStr;
                    });
                } else if (typeof payuResponse === 'string' && payuResponse.includes('html')) {
                    // Fallback if UPI Intent S2S is not enabled or returns HTML
                    alert('UPI Intent S2S returned checkout HTML. Falling back to PayU Hosted Checkout.');
                    document.write(payuResponse);
                } else {
                    alert('Failed to generate Intent QR. Check JSON response for details.');
                }
            } else {
                alert('Failed to generate QR mandate.');
            }
        } catch (error) {
            console.error('Error:', error);
            alert('An error occurred. Check console.');
        } finally {
            qrBtn.disabled = false;
            qrBtn.textContent = 'Generate QR Mandate';
        }
    });

    const upiCollectBtn = document.getElementById('upiCollectBtn');

    upiCollectBtn.addEventListener('click', async (e) => {
        e.preventDefault();
        
        upiCollectBtn.disabled = true;
        upiCollectBtn.textContent = 'Processing...';

        const payload = {
            amount: document.getElementById('amount').value,
            productinfo: document.getElementById('productinfo').value,
            firstname: document.getElementById('firstname').value,
            email: document.getElementById('email').value,
            phone: document.getElementById('phone').value,
            vpa: document.getElementById('vpa').value,
            siDetails: {
                billingAmount: document.getElementById('billingAmount').value,
                billingCurrency: 'INR',
                billingCycle: document.getElementById('billingCycle').value,
                billingInterval: 1,
                paymentStartDate: document.getElementById('paymentStartDate').value,
                paymentEndDate: document.getElementById('paymentEndDate').value,
            }
        };

        try {
            const response = await fetch(endpoints.generateUpiCollect, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });

            const result = await response.json();
            
            if (result.success) {
                responseContainer.classList.remove('hidden');
                jsonResponse.textContent = JSON.stringify(result.data, null, 2);
                
                const payuResponse = result.data;
                if (typeof payuResponse === 'string' && payuResponse.includes('html')) {
                    alert('S2S UPI Collect is not enabled. Falling back to Hosted Checkout.');
                    document.write(payuResponse);
                } else {
                    document.getElementById('qrCodeContainer').style.display = 'block';
                    document.getElementById('qrIntentText').textContent = "UPI Collect Request Sent to " + payload.vpa;
                    document.getElementById('qrImage').style.display = 'none'; // hide QR for collect
                }
            } else {
                alert('Failed to setup mandate via UPI ID.');
            }
        } catch (error) {
            console.error('Error:', error);
            alert('An error occurred. Check console.');
        } finally {
            upiCollectBtn.disabled = false;
            upiCollectBtn.textContent = 'Setup Mandate via UPI ID';
        }
    });
});

/**
 * Creates a hidden form dynamically and submits it to PayU
 * @param {Object} payuData The payload generated by the backend
 */
function submitPayUForm(payuData) {
    const form = document.createElement('form');
    form.method = 'POST';
    form.action = payuData.url;
    
    // Add all keys from payuData as hidden inputs except 'url'
    for (const key in payuData) {
        if (key !== 'url' && payuData.hasOwnProperty(key)) {
            const input = document.createElement('input');
            input.type = 'hidden';
            input.name = key;
            input.value = payuData[key];
            form.appendChild(input);
        }
    }

    document.body.appendChild(form);
    
    // Small delay to allow user to see the JSON payload before redirect
    setTimeout(() => {
        form.submit();
    }, 2000);
}
